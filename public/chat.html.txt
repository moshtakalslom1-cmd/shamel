  <script>
    let chatState = {
      currentUser: null,
      activePeer: '',
      activeAd: null,
      conversations: [],
      messages: [],
      pollTimer: null
    };

    function escapeHtml(text) {
      return String(text || '')
        .replace(/&/g, '&amp;')
        .replace(/</g, '&lt;')
        .replace(/>/g, '&gt;')
        .replace(/"/g, '&quot;');
    }

    function formatTime(ts) {
      const d = ts ? new Date(ts) : new Date();
      if (Number.isNaN(d.getTime())) return '';
      return d.toLocaleTimeString('ar-SY', { hour: '2-digit', minute: '2-digit' });
    }

    function localStorageKey() {
      const u = (chatState.currentUser && chatState.currentUser.username) ? chatState.currentUser.username : 'guest';
      return `rakhis_chat_msgs_${u}`;
    }

    function loadLocalMessages() {
      try {
        const raw = localStorage.getItem(localStorageKey());
        return raw ? JSON.parse(raw) : [];
      } catch (e) {
        return [];
      }
    }

    function saveLocalMessages(list) {
      try {
        localStorage.setItem(localStorageKey(), JSON.stringify(list));
      } catch (e) {}
    }

    window.onload = async () => {
      const savedUser = localStorage.getItem('rakhis_user');
      if (!savedUser) {
        window.location.href = '/';
        return;
      }
      try {
        chatState.currentUser = JSON.parse(savedUser);
      } catch (e) {
        window.location.href = '/';
        return;
      }

      document.getElementById('activeUserBadge').textContent = `👤 ${chatState.currentUser.username}`;

      const params = new URLSearchParams(window.location.search);
      const peerParam = (params.get('to') || '').trim();
      const adIdParam = (params.get('adId') || '').trim();

      if (peerParam && peerParam.toLowerCase() !== chatState.currentUser.username.toLowerCase()) {
        chatState.activePeer = peerParam;
      }

      if (adIdParam) {
        await loadAdContext(adIdParam);
      }

      await refreshChatData();
      chatState.pollTimer = setInterval(refreshChatData, 3000);
    };

    function toggleMobileConversations() {
      const el = document.getElementById('conversationsContainer');
      el.classList.toggle('mobile-open');
    }

    async function loadAdContext(adId) {
      try {
        const res = await fetch(`/api/ads/${encodeURIComponent(adId)}`, { cache: 'no-store' });
        if (!res.ok) return;
        const ad = await res.json();
        chatState.activeAd = ad;
        if (!chatState.activePeer && ad.ownerUsername && ad.ownerUsername !== chatState.currentUser.username) {
          chatState.activePeer = ad.ownerUsername;
        }
        const rawPrice = String(ad.price || '').replace(/[^\d]/g, '');
        const formattedPrice = rawPrice ? Number(rawPrice).toLocaleString('en-US') : '0';
        const banner = document.getElementById('adContextBanner');
        banner.innerHTML = `
          <span>إعلان: <strong>${escapeHtml(ad.title)}</strong></span>
          <span style="color: var(--green); font-weight: 900;">${formattedPrice} ل.س</span>
        `;
        banner.style.display = 'flex';
      } catch (e) {}
    }

    async function refreshChatData() {
      const me = chatState.currentUser.username;
      let serverMsgs = [];
      let serverPeers = [];

      try {
        const res = await fetch(`/api/messages?user=${encodeURIComponent(me)}`, { cache: 'no-store' });
        if (res.ok) {
          const data = await res.json();
          if (Array.isArray(data)) {
            serverMsgs = data;
          } else if (data && Array.isArray(data.messages)) {
            serverMsgs = data.messages;
            if (Array.isArray(data.conversations)) serverPeers = data.conversations;
          }
        }
      } catch (e) {}

      const localMsgs = loadLocalMessages();
      const mergedMap = new Map();
      [...serverMsgs, ...localMsgs].forEach((m, idx) => {
        const key = m.id || `${m.from || m.sender}_${m.to || m.receiver}_${m.timestamp || m.time || idx}_${m.text || m.message || ''}`;
        mergedMap.set(key, {
          id: m.id || key,
          from: m.from || m.sender || '',
          to: m.to || m.receiver || '',
          text: m.text || m.message || '',
          adId: m.adId || '',
          timestamp: m.timestamp || m.time || Date.now()
        });
      });

      chatState.messages = Array.from(mergedMap.values()).sort((a, b) => new Date(a.timestamp) - new Date(b.timestamp));

      const peerMap = new Map();
      serverPeers.forEach(p => {
        const uname = typeof p === 'string' ? p : (p.username || p.peer);
        if (uname && uname.toLowerCase() !== me.toLowerCase()) {
          peerMap.set(uname, {
            username: uname,
            lastMessage: p.lastMessage || '',
            avatar: p.avatar || '',
            lastTime: p.lastTime || ''
          });
        }
      });

      chatState.messages.forEach(m => {
        const isFromMe = String(m.from).toLowerCase() === me.toLowerCase();
        const isToMe = String(m.to).toLowerCase() === me.toLowerCase();
        if (!isFromMe && !isToMe) return;
        const peer = isFromMe ? m.to : m.from;
        if (!peer || peer.toLowerCase() === me.toLowerCase()) return;
        const prev = peerMap.get(peer) || { username: peer, lastMessage: '', avatar: '', lastTime: '' };
        prev.lastMessage = m.text;
        prev.lastTime = formatTime(m.timestamp);
        peerMap.set(peer, prev);
      });

      if (chatState.activePeer && chatState.activePeer.toLowerCase() !== me.toLowerCase()) {
        if (!peerMap.has(chatState.activePeer)) {
          peerMap.set(chatState.activePeer, {
            username: chatState.activePeer,
            lastMessage: 'بدء محادثة جديدة',
            avatar: '',
            lastTime: ''
          });
        }
      }

      chatState.conversations = Array.from(peerMap.values());
      if (!chatState.activePeer && chatState.conversations.length > 0) {
        chatState.activePeer = chatState.conversations[0].username;
      }

      renderConversationsList();
      renderActivePeerHeader();
      renderMessages();
    }

    function renderConversationsList() {
      const container = document.getElementById('conversationsItemsList');
      const q = (document.getElementById('convSearchInput').value || '').trim().toLowerCase();
      const list = chatState.conversations.filter(c => !q || c.username.toLowerCase().includes(q));

      if (list.length === 0) {
        container.innerHTML = '<div style="padding:20px; text-align:center; color:var(--muted); font-size:0.85rem;">لا توجد محادثات بعد. راسل أي معلن من صفحة الإعلانات.</div>';
        return;
      }

      container.innerHTML = list.map(c => {
        const active = c.username.toLowerCase() === (chatState.activePeer || '').toLowerCase();
        const avatarHtml = c.avatar
          ? `<img src="${escapeHtml(c.avatar)}" class="peer-avatar" alt="${escapeHtml(c.username)}">`
          : `<div class="peer-avatar">${escapeHtml(c.username.charAt(0).toUpperCase())}</div>`;

        return `
          <div class="conversation-item ${active ? 'active' : ''}" onclick="switchConversation('${escapeHtml(c.username)}')">
            ${avatarHtml}
            <div style="flex-grow: 1; min-width: 0;">
              <div style="font-weight: 800; font-size: 0.9rem; display: flex; justify-content: space-between; gap: 6px;">
                <span style="overflow: hidden; text-overflow: ellipsis; white-space: nowrap;">${escapeHtml(c.username)}</span>
                <span style="font-size: 0.7rem; color: var(--muted); font-weight: 600;">${escapeHtml(c.lastTime || '')}</span>
              </div>
              <div style="font-size: 0.76rem; color: var(--muted); white-space: nowrap; overflow: hidden; text-overflow: ellipsis;">
                ${escapeHtml(c.lastMessage || 'اضغط لفتح المحادثة')}
              </div>
            </div>
          </div>
        `;
      }).join('');
    }

    function renderActivePeerHeader() {
      const label = document.getElementById('activePeerLabel');
      const status = document.getElementById('activePeerStatus');
      if (!chatState.activePeer) {
        label.textContent = 'اختر شخصاً لبدء المراسلة';
        status.textContent = '';
        return;
      }
      label.textContent = `💬 ${chatState.activePeer}`;
      status.textContent = 'محادثة مباشرة';
    }

    function switchConversation(peerUsername) {
      chatState.activePeer = peerUsername;
      const listEl = document.getElementById('conversationsContainer');
      listEl.classList.remove('mobile-open');
      renderConversationsList();
      renderActivePeerHeader();
      renderMessages();
    }

    function renderMessages() {
      const viewPort = document.getElementById('messagesViewport');
      if (!chatState.activePeer) {
        viewPort.innerHTML = '<div style="text-align: center; color: var(--muted); margin-top: 40px;">اختر محادثة من قائمة الأشخاص الذين راسلتهم.</div>';
        return;
      }

      const me = chatState.currentUser.username.toLowerCase();
      const peer = chatState.activePeer.toLowerCase();

      const thread = chatState.messages.filter(m => {
        const f = String(m.from || '').toLowerCase();
        const t = String(m.to || '').toLowerCase();
        return (f === me && t === peer) || (f === peer && t === me);
      });

      if (thread.length === 0) {
        viewPort.innerHTML = `<div style="text-align: center; color: var(--muted); margin-top: 40px;">ابدأ المحادثة مع ${escapeHtml(chatState.activePeer)} الآن.</div>`;
        return;
      }

      viewPort.innerHTML = thread.map(m => {
        const isSelf = String(m.from || '').toLowerCase() === me;
        return `
          <div class="message-bubble ${isSelf ? 'message-to' : 'message-from'}">
            <div>${escapeHtml(m.text)}</div>
            <span class="message-time">${escapeHtml(formatTime(m.timestamp))}</span>
          </div>
        `;
      }).join('');

      viewPort.scrollTop = viewPort.scrollHeight;
    }

    async function sendChatMessage() {
      const input = document.getElementById('messageInput');
      const text = (input.value || '').trim();
      if (!text) return;

      if (!chatState.activePeer) {
        alert('اختر مستخدماً أولاً من قائمة المحادثات');
        return;
      }

      const payload = {
        id: `${Date.now()}_${Math.random().toString(36).slice(2, 7)}`,
        from: chatState.currentUser.username,
        to: chatState.activePeer,
        sender: chatState.currentUser.username,
        receiver: chatState.activePeer,
        text,
        message: text,
        adId: chatState.activeAd ? chatState.activeAd.id : '',
        timestamp: new Date().toISOString()
      };

      const localList = loadLocalMessages();
      localList.push(payload);
      saveLocalMessages(localList);

      input.value = '';
      await refreshChatData();

      try {
        await fetch('/api/messages', {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify(payload)
        });
      } catch (e) {}
    }
  </script>
